---
title: "Add the messenger to your app"
description: "Install the dimmi messenger in a React, Next.js, Vue, Nuxt, Angular, Svelte or plain JavaScript app with a @dimmi package or a script tag."
---
import { TabItem, Tabs } from "~docs/components/mdx";

Add one package, once, at the root of your app. `<app_id>` is the messenger's app ID (`wgt_...`) from **Settings → Messengers**, and a self-hosted server replaces `https://trydimmi.com` with its own address.

<Tabs>
<TabItem label="React">

```bash
npm install @dimmi/react
```

```tsx title="src/main.tsx"
import { DimmiProvider } from "@dimmi/react";
import { createRoot } from "react-dom/client";

import App from "./App";

createRoot(document.getElementById("root")!).render(
  <DimmiProvider appId="<app_id>" host="https://trydimmi.com">
    <App />
  </DimmiProvider>,
);
```

</TabItem>
<TabItem label="Next.js">

```bash
npm install @dimmi/nextjs
```

```tsx title="app/layout.tsx"
import { Dimmi } from "@dimmi/nextjs";
import type { ReactNode } from "react";

export default function RootLayout({ children }: { children: ReactNode }) {
  return (
    <html lang="en">
      <body>
        {children}
        <Dimmi appId="<app_id>" host="https://trydimmi.com" />
      </body>
    </html>
  );
}
```

<TabItem label="Vue">

```bash
npm install @dimmi/vue
```

```ts title="src/main.ts"
import { createDimmi } from "@dimmi/vue";
import { createApp } from "vue";

import App from "./App.vue";

createApp(App)
  .use(createDimmi({ appId: "<app_id>", host: "https://trydimmi.com" }))
```

## Show the unread count

```tsx title="src/HelpButton.tsx"
import { DimmiButton, useUnreadCount } from "@dimmi/react";

export function HelpButton() {
  const unread = useUnreadCount();
  return <DimmiButton>Chat with us ({unread})</DimmiButton>;
}
```

| Package | Unread count |
| --- | --- |
| `@dimmi/react`, `@dimmi/nextjs` | `useUnreadCount()` |
| `@dimmi/vue` | `useUnreadCount()`, a `Ref<number>` |
| `@dimmi/angular` | `DimmiService.unreadCount`, a signal |
| `@dimmi/svelte` | `unreadCount`, a store |
| `@dimmi/browser` | `getDimmi().onUnreadCountChange(fn)` |
| Script tag | `event.detail.count` of the `dimmi:unread` window event |
